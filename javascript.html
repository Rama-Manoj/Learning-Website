Variables store data that can be used and modified throughout your JavaScript code. Understanding variable declarations is fundamental to JavaScript programming.

Variable Declaration Types:

  • const: Cannot be reassigned (preferred)
  • let: Block-scoped, can be reassigned
  • var: Function-scoped, avoid in modern code

Data Types:

  • Primitives: string, number, boolean, undefined, null, symbol
  • Objects: arrays, functions, objects

Best Practices:

  • Use const by default
  • Use let when reassignment is needed
  • Avoid var
  • Use descriptive variable names
// Variable declarations
const greeting = 'Hello';
let name = 'Ada';
const message = `${greeting}, ${name}!`;

// Different data types
const age = 25;                    // number
const isStudent = true;            // boolean
const hobbies = ['reading', 'coding']; // array
const person = {                   // object
  name: 'Alice',
  age: 30
};

// Template literals (string interpolation)
const info = `Name: ${person.name}, Age: ${person.age}`;

// Variable reassignment (only with let)
let counter = 0;
counter = counter + 1;

// Block scope example
if (true) {
  const blockScoped = 'I exist only in this block';
  let alsoBlockScoped = 'Me too';
}
// blockScoped and alsoBlockScoped are not accessible here

Functions are reusable blocks of code that perform specific tasks. They help organize code and avoid repetition.

Function Types:

  • Function Declaration: function name() {}
  • Function Expression: const name = function() {}
  • Arrow Function: const name = () => {}

Key Concepts:

  • Parameters: Input values for the function
  • Return: Output value from the function
  • Scope: Where variables are accessible
  • Hoisting: Function declarations are hoisted

Arrow Functions:

  • Shorter syntax
  • No own this binding
  • Implicit return for single expressions
// Function Declaration
function greet(name) {
  return `Hello, ${name}!`;
}

// Function Expression
const add = function(a, b) {
  return a + b;
};

// Arrow Function
const multiply = (a, b) => a * b;

// Arrow function with block body
const divide = (a, b) => {
  if (b === 0) {
    return 'Cannot divide by zero';
  }
  return a / b;
};

// Function with default parameters
function createUser(name, age = 18, isActive = true) {
  return {
    name,
    age,
    isActive,
    greet: () => `Hi, I'm ${name}`
  };
}

// Higher-order function (takes function as parameter)
function processNumbers(numbers, operation) {
  return numbers.map(operation);
}

// Usage examples
const numbers = [1, 2, 3, 4, 5];
const doubled = processNumbers(numbers, x => x * 2);
const squared = processNumbers(numbers, x => x * x);

// Callback function
setTimeout(() => {
  console.log('This runs after 1 second');
}, 1000);

The Document Object Model (DOM) allows JavaScript to interact with HTML elements, modify content, attributes, and styles dynamically.

Element Selection:

  • document.getElementById(): Select by ID
  • document.querySelector(): CSS selector
  • document.querySelectorAll(): Multiple elements
  • document.getElementsByClassName(): By class

Content Manipulation:

  • textContent: Text content only
  • innerHTML: HTML content
  • innerText: Visible text

Attribute & Style:

  • setAttribute(), getAttribute()
  • style.property: Inline styles
  • classList: CSS classes
// Element selection
const element = document.getElementById('myElement');
const elements = document.querySelectorAll('.my-class');
const firstDiv = document.querySelector('div');

// Content manipulation
element.textContent = 'New text content';
element.innerHTML = 'Bold text';
element.innerText = 'Visible text only';

// Attribute manipulation
element.setAttribute('data-id', '123');
element.setAttribute('class', 'new-class');
const id = element.getAttribute('data-id');

// Style manipulation
element.style.color = 'red';
element.style.fontSize = '20px';
element.style.backgroundColor = '#f0f0f0';

// Class manipulation
element.classList.add('active');
element.classList.remove('inactive');
element.classList.toggle('visible');
element.classList.contains('active'); // returns boolean

// Creating elements
const newDiv = document.createElement('div');
newDiv.textContent = 'New element';
newDiv.className = 'new-element';

// Adding to DOM
element.appendChild(newDiv);
element.insertBefore(newDiv, element.firstChild);

// Removing elements
element.removeChild(newDiv);
newDiv.remove(); // Modern way

// Event handling
element.addEventListener('click', function(event) {
  console.log('Element clicked!');
  event.target.style.color = 'blue';
});

// Form handling
const form = document.querySelector('form');
const input = document.querySelector('input[type="text"]');
const value = input.value; // Get input value
input.value = 'New value'; // Set input value

Events allow JavaScript to respond to user interactions and browser actions, making web pages interactive and dynamic.

Common Event Types:

  • Mouse: click, dblclick, mouseover, mouseout
  • Keyboard: keydown, keyup, keypress
  • Form: submit, change, input, focus, blur
  • Window: load, resize, scroll

Event Handling:

  • addEventListener(): Modern way to add events
  • removeEventListener(): Remove event listeners
  • event.preventDefault(): Prevent default behavior
  • event.stopPropagation(): Stop event bubbling

Event Object:

  • event.target: Element that triggered event
  • event.type: Type of event
  • event.key: Key pressed (keyboard events)
// Basic event listener
const button = document.getElementById('myButton');
button.addEventListener('click', function(event) {
  console.log('Button clicked!');
  console.log('Event type:', event.type);
  console.log('Target element:', event.target);
});

// Mouse events
const element = document.getElementById('myElement');
element.addEventListener('mouseover', function() {
  this.style.backgroundColor = 'lightblue';
});

element.addEventListener('mouseout', function() {
  this.style.backgroundColor = 'white';
});

// Keyboard events
document.addEventListener('keydown', function(event) {
  console.log('Key pressed:', event.key);
  console.log('Key code:', event.keyCode);
  
  // Handle specific keys
  if (event.key === 'Enter') {
    console.log('Enter key pressed!');
  }
});

// Form events
const form = document.querySelector('form');
const input = document.querySelector('input');

form.addEventListener('submit', function(event) {
  event.preventDefault(); // Prevent form submission
  console.log('Form submitted!');
});

input.addEventListener('input', function(event) {
  console.log('Input value:', event.target.value);
});

input.addEventListener('focus', function() {
  this.style.borderColor = 'blue';
});

input.addEventListener('blur', function() {
  this.style.borderColor = 'gray';
});

// Window events
window.addEventListener('load', function() {
  console.log('Page fully loaded!');
});

window.addEventListener('resize', function() {
  console.log('Window resized!');
});

// Event delegation (handle events on parent)
const list = document.getElementById('myList');
list.addEventListener('click', function(event) {
  if (event.target.tagName === 'LI') {
    console.log('List item clicked:', event.target.textContent);
  }
});

// Custom events
const customEvent = new CustomEvent('myCustomEvent', {
  detail: { message: 'Hello from custom event!' }
});

element.addEventListener('myCustomEvent', function(event) {
  console.log(event.detail.message);
});

element.dispatchEvent(customEvent);

Arrays are ordered collections of values that can be accessed by index. They provide powerful methods for manipulation and iteration.

Array Creation:

  • []: Literal syntax
  • new Array(): Constructor
  • Array.from(): From iterable

Common Methods:

  • Modification: push, pop, shift, unshift, splice
  • Search: indexOf, find, includes
  • Iteration: forEach, map, filter, reduce
  • Transformation: join, slice, concat

Array Destructuring:

  • const [first, second] = array
  • const [first, ...rest] = array
// Array creation
const fruits = ['apple', 'banana', 'orange'];
const numbers = [1, 2, 3, 4, 5];
const mixed = [1, 'hello', true, { name: 'Alice' }];

// Array methods
fruits.push('grape');        // Add to end
fruits.pop();                // Remove from end
fruits.unshift('kiwi');      // Add to beginning
fruits.shift();              // Remove from beginning

// Accessing elements
console.log(fruits[0]);      // First element
console.log(fruits.length); // Array length
console.log(fruits[fruits.length - 1]); // Last element

// Iteration methods
fruits.forEach(fruit => {
  console.log(fruit);
});

// Map creates new array
const upperFruits = fruits.map(fruit => fruit.toUpperCase());

// Filter creates new array with condition
const longFruits = fruits.filter(fruit => fruit.length > 5);

// Reduce accumulates values
const sum = numbers.reduce((acc, num) => acc + num, 0);

// Find elements
const found = fruits.find(fruit => fruit.startsWith('a'));
const index = fruits.indexOf('banana');

// Array destructuring
const [first, second, ...rest] = fruits;
console.log(first);  // 'apple'
console.log(rest);   // ['orange', 'grape']

// Spread operator
const newFruits = [...fruits, 'mango'];
const combined = [...fruits, ...numbers];

// Array methods chaining
const result = numbers
  .filter(num => num > 2)
  .map(num => num * 2)
  .reduce((sum, num) => sum + num, 0);

Loops allow you to execute code repeatedly, making it efficient to work with collections and perform repetitive tasks.

Loop Types:

  • for: Traditional loop with counter
  • while: Loop while condition is true
  • do...while: Execute at least once
  • for...of: Iterate over iterables
  • for...in: Iterate over object properties

Loop Control:

  • break: Exit loop immediately
  • continue: Skip current iteration

Modern Alternatives:

  • forEach(): Array iteration
  • map(): Transform array
  • filter(): Filter array
// For loop
for (let i = 0; i < 5; i++) {
  console.log(`Count: ${i}`);
}

// For loop with array
const fruits = ['apple', 'banana', 'orange'];
for (let i = 0; i < fruits.length; i++) {
  console.log(fruits[i]);
}

// While loop
let count = 0;
while (count < 3) {
  console.log(`While count: ${count}`);
  count++;
}

// Do...while loop
let num = 0;
do {
  console.log(`Do-while: ${num}`);
  num++;
} while (num < 3);

// For...of loop (arrays, strings)
for (const fruit of fruits) {
  console.log(fruit);
}

// For...of with strings
for (const char of 'hello') {
  console.log(char);
}

// For...in loop (object properties)
const person = { name: 'Alice', age: 30, city: 'NYC' };
for (const key in person) {
  console.log(`${key}: ${person[key]}`);
}

// Loop control
for (let i = 0; i < 10; i++) {
  if (i === 3) {
    continue; // Skip iteration
  }
  if (i === 7) {
    break; // Exit loop
  }
  console.log(i);
}

// Nested loops
for (let i = 0; i < 3; i++) {
  for (let j = 0; j < 3; j++) {
    console.log(`(${i}, ${j})`);
  }
}

// Modern array methods (preferred)
fruits.forEach(fruit => {
  console.log(fruit);
});

const doubled = fruits.map(fruit => fruit.toUpperCase());
const filtered = fruits.filter(fruit => fruit.length > 5);

Conditional statements allow your code to make decisions and execute different code paths based on different conditions.

Conditional Statements:

  • if: Execute if condition is true
  • else if: Additional conditions
  • else: Default case
  • switch: Multiple value comparisons

Comparison Operators:

  • ==, ===: Equality (use ===)
  • !=, !==: Inequality
  • >, <, >=, <=: Comparison

Logical Operators:

  • &&: AND
  • ||: OR
  • !: NOT
// Basic if statement
const age = 18;
if (age >= 18) {
  console.log('You are an adult');
}

// If...else statement
const temperature = 25;
if (temperature > 30) {
  console.log('It\'s hot!');
} else if (temperature > 20) {
  console.log('It\'s warm');
} else {
  console.log('It\'s cool');
}

// Logical operators
const isStudent = true;
const hasDiscount = false;

if (isStudent || hasDiscount) {
  console.log('You get a discount!');
}

if (isStudent && age >= 16) {
  console.log('You can attend school');
}

// Ternary operator (shorthand if...else)
const status = age >= 18 ? 'adult' : 'minor';
const message = isStudent ? 'Welcome student!' : 'Welcome!';

// Switch statement
const day = 'Monday';
switch (day) {
  case 'Monday':
    console.log('Start of work week');
    break;
  case 'Friday':
    console.log('TGIF!');
    break;
  case 'Saturday':
  case 'Sunday':
    console.log('Weekend!');
    break;
  default:
    console.log('Regular day');
}

// Truthy and falsy values
const name = ''; // falsy
const count = 0; // falsy
const isActive = false; // falsy
const user = null; // falsy

if (name) {
  console.log('Name exists');
} else {
  console.log('No name provided');
}

// Short-circuit evaluation
const username = name || 'Guest';
const isLoggedIn = user && user.isActive;

// Optional chaining (modern)
const userAge = user?.age;
const userName = user?.profile?.name;

Objects are collections of key-value pairs that allow you to store and organize related data and functionality.

Object Creation:

  • Literal: { key: value }
  • Constructor: new Object()
  • Factory: Functions that return objects

Object Properties:

  • object.property: Dot notation
  • object['property']: Bracket notation
  • Object.keys(): Get all keys
  • Object.values(): Get all values

Object Methods:

  • Functions as object properties
  • this keyword refers to object
  • Method shorthand in ES6
// Object literal
const person = {
  name: 'Alice',
  age: 30,
  city: 'New York',
  isStudent: false
};

// Accessing properties
console.log(person.name);        // Dot notation
console.log(person['age']);      // Bracket notation

// Adding properties
person.email = 'alice@example.com';
person['phone'] = '123-456-7890';

// Object methods
const calculator = {
  add: function(a, b) {
    return a + b;
  },
  subtract(a, b) {  // Method shorthand
    return a - b;
  },
  multiply: (a, b) => a * b  // Arrow function
};

// Object with methods using this
const user = {
  name: 'Bob',
  greet() {
    return `Hello, I'm ${this.name}`;
  },
  updateName(newName) {
    this.name = newName;
  }
};

// Object destructuring
const { name, age, city } = person;
const { name: userName, age: userAge } = person; // Rename

// Spread operator with objects
const updatedPerson = { ...person, age: 31 };
const merged = { ...person, ...user };

// Object methods
const keys = Object.keys(person);
const values = Object.values(person);
const entries = Object.entries(person);

// Check if property exists
if ('name' in person) {
  console.log('Name property exists');
}

if (person.hasOwnProperty('age')) {
  console.log('Age property exists');
}

// Object.assign (alternative to spread)
const newPerson = Object.assign({}, person, { age: 32 });

// Computed property names
const propName = 'dynamicProp';
const dynamicObj = {
  [propName]: 'value',
  [`${propName}2`]: 'another value'
};

Async/await provides a clean way to handle asynchronous operations, making code more readable than traditional callbacks or promises.

Async/Await Concepts:

  • async: Function returns a Promise
  • await: Wait for Promise to resolve
  • try/catch: Handle errors
  • Promise.all(): Wait for multiple promises

Common Use Cases:

  • API calls with fetch
  • File operations
  • Database queries
  • Timers and delays

Error Handling:

  • Use try/catch blocks
  • Handle rejected promises
  • Provide fallback values
// Basic async function
async function fetchData() {
  try {
    const response = await fetch('https://api.example.com/data');
    const data = await response.json();
    return data;
  } catch (error) {
    console.error('Error fetching data:', error);
    return null;
  }
}

// Async function with delay
async function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function processWithDelay() {
  console.log('Starting...');
  await delay(1000);
  console.log('After 1 second');
  await delay(1000);
  console.log('After 2 seconds');
}

// Multiple async operations
async function fetchMultipleData() {
  try {
    const [users, posts, comments] = await Promise.all([
      fetch('/api/users').then(res => res.json()),
      fetch('/api/posts').then(res => res.json()),
      fetch('/api/comments').then(res => res.json())
    ]);
    
    return { users, posts, comments };
  } catch (error) {
    console.error('Error fetching multiple data:', error);
  }
}

// Async function with error handling
async function getUserData(userId) {
  try {
    const response = await fetch(`/api/users/${userId}`);
    
    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`);
    }
    
    const userData = await response.json();
    return userData;
  } catch (error) {
    console.error('Failed to get user data:', error);
    return { error: error.message };
  }
}

// Async function in event handler
document.getElementById('loadButton').addEventListener('click', async function() {
  this.textContent = 'Loading...';
  this.disabled = true;
  
  try {
    const data = await fetchData();
    this.textContent = 'Data loaded!';
    console.log(data);
  } catch (error) {
    this.textContent = 'Error loading data';
  } finally {
    this.disabled = false;
  }
});

// Async generator
async function* asyncGenerator() {
  for (let i = 0; i < 3; i++) {
    await delay(500);
    yield i;
  }
}

// Using async generator
async function useAsyncGenerator() {
  for await (const value of asyncGenerator()) {
    console.log(value);
  }
}

Error handling allows your code to gracefully handle unexpected situations and provide meaningful feedback to users.

Error Handling Methods:

  • try/catch: Handle synchronous errors
  • try/catch with async: Handle async errors
  • throw: Create custom errors
  • finally: Cleanup code

Error Types:

  • Error: Generic error
  • TypeError: Wrong data type
  • ReferenceError: Undefined variable
  • SyntaxError: Code syntax issues

Best Practices:

  • Handle errors gracefully
  • Provide meaningful error messages
  • Log errors for debugging
  • Use specific error types
// Basic try/catch
try {
  const result = riskyOperation();
  console.log('Success:', result);
} catch (error) {
  console.error('Error occurred:', error.message);
}

// Custom error throwing
function divide(a, b) {
  if (b === 0) {
    throw new Error('Division by zero is not allowed');
  }
  return a / b;
}

// Try/catch with custom error
try {
  const result = divide(10, 0);
  console.log(result);
} catch (error) {
  console.error('Division error:', error.message);
}

// Async error handling
async function fetchData() {
  try {
    const response = await fetch('/api/data');
    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`);
    }
    return await response.json();
  } catch (error) {
    console.error('Fetch error:', error.message);
    return null;
  }
}

// Finally block
function processFile(filename) {
  let file = null;
  try {
    file = openFile(filename);
    return processFileContent(file);
  } catch (error) {
    console.error('File processing error:', error.message);
    return null;
  } finally {
    if (file) {
      closeFile(file);
    }
  }
}

// Error types
try {
  const obj = null;
  obj.property; // TypeError
} catch (error) {
  if (error instanceof TypeError) {
    console.log('Type error:', error.message);
  }
}

// Custom error class
class ValidationError extends Error {
  constructor(message, field) {
    super(message);
    this.name = 'ValidationError';
    this.field = field;
  }
}

function validateUser(user) {
  if (!user.name) {
    throw new ValidationError('Name is required', 'name');
  }
  if (!user.email) {
    throw new ValidationError('Email is required', 'email');
  }
}

// Global error handling
window.addEventListener('error', function(event) {
  console.error('Global error:', event.error);
});

// Promise error handling
fetch('/api/data')
  .then(response => response.json())
  .then(data => console.log(data))
  .catch(error => console.error('Promise error:', error));

// Unhandled promise rejection
window.addEventListener('unhandledrejection', function(event) {
  console.error('Unhandled promise rejection:', event.reason);
});

ES6 modules allow you to organize code into separate files, making it easier to maintain and reuse code across different parts of your application.

Module Types:

  • Named exports: Multiple exports per module
  • Default export: Single main export
  • Mixed exports: Both named and default

Import/Export Syntax:

  • export: Export from module
  • import: Import into module
  • export default: Default export
  • import ... as: Rename imports

Module Benefits:

  • Code organization
  • Reusability
  • Dependency management
  • Tree shaking (dead code elimination)
// math.js - Named exports
export const add = (a, b) => a + b;
export const subtract = (a, b) => a - b;
export const multiply = (a, b) => a * b;
export const divide = (a, b) => a / b;

// utils.js - Default export
const formatDate = (date) => {
  return date.toLocaleDateString();
};

export default formatDate;

// user.js - Mixed exports
export const createUser = (name, email) => ({
  name,
  email,
  id: Date.now()
});

export const validateUser = (user) => {
  return user.name && user.email;
};

const UserService = {
  createUser,
  validateUser
};

export default UserService;

// main.js - Importing modules
import { add, subtract } from './math.js';
import formatDate from './utils.js';
import UserService, { createUser } from './user.js';

// Using imported functions
const result = add(5, 3);
const formatted = formatDate(new Date());
const user = createUser('Alice', 'alice@example.com');

// Import all exports
import * as Math from './math.js';
const sum = Math.add(10, 5);

// Import with renaming
import { add as addition } from './math.js';
import { createUser as newUser } from './user.js';

// Dynamic imports
async function loadModule() {
  const module = await import('./math.js');
  return module.add(1, 2);
}

// Conditional imports
if (condition) {
  const { specialFunction } = await import('./special.js');
  specialFunction();
}

// HTML script tag with modules
// <script type="module" src="main.js"></script>

// Re-exporting
export { add, subtract } from './math.js';
export { default as UserService } from './user.js';
